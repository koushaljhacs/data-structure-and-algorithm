## 1. What are Route Parameters?
Route parameters are named URL segments that capture dynamic values from the URL path.
They are defined using a colon (:) prefix in the route path.

Syntax: /path/:parameterName

Example:
app.get('/user/:id', (req, res) => {
    const userId = req.params.id;
    res.send(`User ID: ${userId}`);
});

## 2. Basic Syntax and Usage

### Your Original Code Explained:
let express = require('express');
let app = express();

// Route 1: Product route with parameter
app.get('/product/:id', (req, res) => {
    console.log(req.params);        // Logs: { id: 'value' }
    res.send("this is a product route");
})

// Route 2: Todo route with parameter
app.get('/todo/:id', (req, res) => {
    console.log(req.params.id);     // Logs only the value
    res.send("this is a todo route");
})

app.listen(3000, () => {
    console.log("app is running on port 3000...");
})

### What happens when you run:
1. Start server: node app.js
2. Visit: http://localhost:3000/product/123
3. Console shows: [Object: null prototype] { id: '123' }
4. Browser shows: "this is a product route"

### URL Matching Patterns:

URL Pattern     | Test URL           | req.params        | req.params.id
----------------|--------------------|-------------------|---------------
/product/:id    | /product/123       | { id: '123' }     | '123'
/product/:id    | /product/apple     | { id: 'apple' }   | 'apple'
/product/:id    | /product/abc-123   | { id: 'abc-123'}  | 'abc-123'
/todo/:id       | /todo/456          | { id: '456' }     | '456'

## 3. Accessing Parameter Values

### Method 1: Complete params object
app.get('/product/:id', (req, res) => {
    console.log(req.params);  // { id: '123' }
    console.log(typeof req.params);  // 'object'
    
    // Note: [Object: null prototype] means it's a plain object
    // without standard Object methods (for performance)
});

### Method 2: Specific parameter (Recommended)
app.get('/product/:id', (req, res) => {
    const productId = req.params.id;
    console.log(productId);  // '123'
    console.log(typeof productId);  // 'string' (ALWAYS string!)
    
    // Convert to number if needed
    const numericId = parseInt(productId);
    const floatId = parseFloat(productId);
    
    res.send(`Product ID: ${numericId}`);
});

### Method 3: Destructuring (ES6)
app.get('/product/:id', (req, res) => {
    const { id } = req.params;  // Destructure
    console.log(id);  // Direct access
    res.send(`ID: ${id}`);
});

## 4. Multiple Parameters

### Multiple parameters in single route:
app.get('/category/:categoryId/product/:productId', (req, res) => {
    console.log(req.params);
    // For: /category/electronics/product/789
    // Output: { categoryId: 'electronics', productId: '789' }
    
    const { categoryId, productId } = req.params;
    res.send(`Category: ${categoryId}, Product: ${productId}`);
});

### Three or more parameters:
app.get('/user/:userId/posts/:postId/comments/:commentId', (req, res) => {
    const { userId, postId, commentId } = req.params;
    res.send(`User ${userId}, Post ${postId}, Comment ${commentId}`);
});

## 5. Optional Parameters

### Using question mark (?):
app.get('/books/:id?', (req, res) => {
    if (req.params.id) {
        res.send(`Book ID: ${req.params.id}`);
    } else {
        res.send('Showing all books');
    }
});

// URL: /books/123  → Shows book 123
// URL: /books      → Shows all books

### Multiple optional parameters:
app.get('/search/:category?/:keyword?', (req, res) => {
    const { category, keyword } = req.params;
    
    if (category && keyword) {
        res.send(`Searching ${keyword} in ${category}`);
    } else if (category) {
        res.send(`Browsing category: ${category}`);
    } else if (keyword) {
        res.send(`Searching for: ${keyword}`);
    } else {
        res.send('Showing all items');
    }
});

## 6. Parameter Constraints with Regex

### Only match numeric IDs:
app.get('/user/:id(\\d+)', (req, res) => {
    res.send(`User ID (numeric only): ${req.params.id}`);
});
// Matches: /user/123
// Rejects: /user/abc

### Only match alphabetical names:
app.get('/profile/:name([a-zA-Z]+)', (req, res) => {
    res.send(`Profile: ${req.params.name}`);
});
// Matches: /profile/john
// Rejects: /profile/john123

### Complex constraints:
app.get('/product/:id(\\d+)-:slug([a-z-]+)', (req, res) => {
    // Matches: /product/123-laptop-pro
    // params: { id: '123', slug: 'laptop-pro' }
    res.send(`Product ${req.params.id}: ${req.params.slug}`);
});

### Date format constraint:
app.get('/blog/:year(\\d{4})/:month(\\d{2})/:day(\\d{2})', (req, res) => {
    const { year, month, day } = req.params;
    res.send(`Blog from ${day}/${month}/${year}`);
});

## 7. Real-World Examples

### Example 1: E-commerce Product Page
app.get('/products/:productId', (req, res) => {
    const productId = parseInt(req.params.productId);
    
    // Database lookup (pseudo-code)
    const product = products.find(p => p.id === productId);
    
    if (product) {
        res.json({
            success: true,
            data: {
                id: product.id,
                name: product.name,
                price: product.price,
                category: product.category
            }
        });
    } else {
        res.status(404).json({
            success: false,
            message: `Product with ID ${productId} not found`
        });
    }
});

### Example 2: Blog System
app.get('/:username/posts/:postSlug', (req, res) => {
    const { username, postSlug } = req.params;
    
    res.send(`
        <!DOCTYPE html>
        <html>
        <head>
            <title>${postSlug} - by ${username}</title>
        </head>
        <body>
            <h1>${postSlug.replace(/-/g, ' ')}</h1>
            <p>Author: ${username}</p>
            <p>URL Parameters captured:</p>
            <ul>
                <li>Username: ${username}</li>
                <li>Post Slug: ${postSlug}</li>
            </ul>
        </body>
        </html>
    `);
});

### Example 3: API Versioning
app.get('/api/v:version/users/:userId', (req, res) => {
    const { version, userId } = req.params;
    
    if (version === '1') {
        // Old API format
        res.json({ userId, name: "User from v1" });
    } else if (version === '2') {
        // New API format
        res.json({ 
            data: { 
                id: userId, 
                attributes: { name: "User from v2" } 
            } 
        });
    } else {
        res.status(400).json({ error: 'Unsupported API version' });
    }
});

## 8. Common Patterns and Best Practices

### 1. Always Validate and Sanitize
app.get('/product/:id', (req, res) => {
    const productId = parseInt(req.params.id);
    
    // Validation
    if (isNaN(productId)) {
        return res.status(400).json({ error: 'Invalid product ID' });
    }
    
    if (productId <= 0) {
        return res.status(400).json({ error: 'Product ID must be positive' });
    }
    
    // Sanitization (if needed)
    const safeProductId = productId;
    
    // Continue with processing...
});

### 2. Use Consistent Naming Conventions
// Good - consistent
app.get('/users/:userId', getUser);
app.get('/products/:productId', getProduct);
app.get('/orders/:orderId', getOrder);

// Avoid - inconsistent
app.get('/user/:id', getUser);
app.get('/products/:prodId', getProduct);
app.get('/orders/:order', getOrder);

### 3. Route Order Matters
// WRONG - Specific route after parameterized route
app.get('/product/:id', (req, res) => {
    res.send('Generic product route');
});

app.get('/product/new', (req, res) => {
    res.send('Create new product'); // This won't be reached!
});

// CORRECT - Specific routes first
app.get('/product/new', (req, res) => {
    res.send('Create new product');
});

app.get('/product/:id', (req, res) => {
    res.send('Generic product route');
});

### 4. Use Router for Organizing
// In routes/products.js
const router = require('express').Router();

router.get('/:id', (req, res) => {
    res.send(`Product ${req.params.id}`);
});

router.get('/:id/reviews', (req, res) => {
    res.send(`Reviews for product ${req.params.id}`);
});

module.exports = router;

// In main app.js
const productRoutes = require('./routes/products');
app.use('/products', productRoutes);

## 9. Common Mistakes to Avoid

### Mistake 1: Forgetting parameters are strings
// WRONG
app.get('/product/:id', (req, res) => {
    if (req.params.id === 123) {  // '123' !== 123
        // Never executes
    }
});

// CORRECT
app.get('/product/:id', (req, res) => {
    const id = parseInt(req.params.id);
    if (id === 123) {
        // Works correctly
    }
});

### Mistake 2: Not handling missing parameters
// Add validation
app.get('/user/:id', (req, res) => {
    if (!req.params.id || req.params.id.trim() === '') {
        return res.status(400).send('User ID is required');
    }
    // Process...
});

### Mistake 3: Overly broad route patterns
// Too broad - catches everything
app.get('/:anything', (req, res) => {
    // This will catch ALL GET requests!
});

// Better - be specific
app.get('/api/:resource/:id', (req, res) => {
    // Only catches /api/*/* patterns
});
